-- ─────────────────────────────────────────────────────────────
-- Network layer  (ReplicatedStorage.Network.<Service>.<RE|RF|RP>.<Name>)
--   service = "" หมายถึง Network.RE / Network.RF โดยตรง (เช่น BuyUpgrade)
-- ─────────────────────────────────────────────────────────────
local RemoteCache = {}
local function getRemote(service, name)
    if not Network then return nil end
    local key = service .. "." .. name
    local hit = RemoteCache[key]
    if hit and hit.Parent then return hit end
    local root = (service == "") and Network or Network:FindFirstChild(service)
    local found
    if root then
        for _, kind in ipairs({ "RE", "RF", "RP" }) do
            local f = root:FindFirstChild(kind)
            local r = f and f:FindFirstChild(name)
            if r then found = r break end
        end
    end
    RemoteCache[key] = found
    return found
end

local function send(service, name, ...)
    local r = getRemote(service, name)
    if not r then
        Stats.Errors = Stats.Errors + 1
        dbg("remote not found:", service, name)
        return false
    end
    local args = table.pack(...)
    Stats.Calls = Stats.Calls + 1
    if r:IsA("RemoteEvent") then
        local ok, err = pcall(function() r:FireServer(table.unpack(args, 1, args.n)) end)
        if not ok then Stats.Errors = Stats.Errors + 1 dbg(err) end
        return ok
    elseif r:IsA("RemoteFunction") then
        local ok, res = pcall(function() return r:InvokeServer(table.unpack(args, 1, args.n)) end)
        if not ok then Stats.Errors = Stats.Errors + 1 dbg(res) return false end
        return true, res
    end
    return false
end

-- ─────────────────────────────────────────────────────────────
-- Learn Mode: จำ argument จริงที่เกมส่งตอนผู้เล่นกดเอง
-- ─────────────────────────────────────────────────────────────
local Learned, LearnOrder = {}, {}
local LearnLog = {}
local LearnStatus = "ยังไม่เริ่ม"
local LearnHooked = false

local IGNORE_REMOTE = { Data = true, Cache = true, Stock = true, ActiveWeather = true, EquippedUnit = true, PlotId = true }

local function keyOf(remote)
    local svc = remote.Parent and remote.Parent.Parent
    local sname = (svc and svc ~= Network) and svc.Name or "Network"
    return sname .. "." .. remote.Name, (svc and svc ~= Network) and svc.Name or ""
end

local function describeArg(v)
    local t = typeof(v)
    if t == "string" then return '"' .. v:sub(1, 24) .. '"' end
    if t == "Instance" then return "<" .. v.ClassName .. ":" .. v.Name .. ">" end
    if t == "table" then return "{table}" end
    return tostring(v)
end

local function record(remote, args, method)
    local key, svcName = keyOf(remote)
    local entry = Learned[key]
    if not entry then
        entry = { remote = remote, service = svcName, name = remote.Name, count = 0 }
        Learned[key] = entry
        LearnOrder[#LearnOrder + 1] = key
        table.sort(LearnOrder)
    end
    entry.args = args
    entry.count = entry.count + 1
    entry.time = os.clock()
    local parts = {}
    for i = 1, math.min(args.n, 5) do parts[#parts + 1] = describeArg(args[i]) end
    table.insert(LearnLog, 1, ("%s(%s)"):format(key, table.concat(parts, ", ")))
    if #LearnLog > 8 then table.remove(LearnLog) end
end

local function installLearner()
    if LearnHooked then return true end
    if not (hookmetamethod and getnamecallmethod) then
        LearnStatus = "executor ไม่รองรับ hookmetamethod"
        return false
    end
    local old
    local function handler(self, ...)
        local m = getnamecallmethod()
        if (m == "FireServer" or m == "InvokeServer") and typeof(self) == "Instance"
            and not (checkcaller and checkcaller()) then
            local args = table.pack(...)
            pcall(function()
                if Network and self:IsDescendantOf(Network) and not IGNORE_REMOTE[self.Name] then
                    record(self, args, m)
                end
            end)
        end
        return old(self, ...)
    end
    local ok = pcall(function()
        old = hookmetamethod(game, "__namecall", newcclosure and newcclosure(handler) or handler)
    end)
    if ok and old then
        LearnHooked = true
        LearnStatus = "กำลังจำค่า (กดในเกมได้เลย)"
        return true
    end
    LearnStatus = "ติดตั้ง hook ไม่สำเร็จ"
    return false
end

local function replay(key)
    local e = Learned[key]
    if not (e and e.remote and e.remote.Parent) then return false end
    local args = e.args or table.pack()
    Stats.Calls = Stats.Calls + 1
    local r = e.remote
    if r:IsA("RemoteEvent") then
        return pcall(function() r:FireServer(table.unpack(args, 1, args.n)) end)
    elseif r:IsA("RemoteFunction") then
        return pcall(function() return r:InvokeServer(table.unpack(args, 1, args.n)) end)
    end
    return false
end

-- ใช้ค่าที่เรียนรู้ไว้ ถ้ายังไม่เคยเรียนรู้ ค่อยยิงแบบไม่มี argument
local function sendSmart(service, name, ...)
    local key = (service == "" and "Network" or service) .. "." .. name
    if Learned[key] then return replay(key) end
    return send(service, name, ...)
end

-- ─────────────────────────────────────────────────────────────
-- Loop manager
-- ─────────────────────────────────────────────────────────────
local Loops = {}
local Alive = true
local function startLoop(flag, getDelay, fn)
    if Loops[flag] then return end
    local token = {}
    Loops[flag] = token
    task.spawn(function()
        while Alive and Loops[flag] == token and State[flag] do
            local ok, err = pcall(fn)
            if not ok then Stats.Errors = Stats.Errors + 1 dbg(flag, err) end
            task.wait(math.max(0.1, tonumber(getDelay()) or 1))
        end
        if Loops[flag] == token then Loops[flag] = nil end
    end)
end

local function setLoop(flag, v, getDelay, fn)
    State[flag] = v and true or false
    markDirty()
    if v then startLoop(flag, getDelay, fn) else Loops[flag] = nil end
end

        local function drag(handle)
            local dragging, startIn, startPos
            handle.InputBegan:Connect(function(input)
                if isPointer(input) then
                    dragging, startIn, startPos = true, input.Position, main.Position
                    input.Changed:Connect(function()
                        if input.UserInputState == Enum.UserInputState.End then dragging = false end
                    end)
                end
            end)
            track(UIS.InputChanged:Connect(function(input)
                if dragging and isMove(input) then
                    local d = input.Position - startIn
                    main.Position = UDim2.fromOffset(startPos.X.Offset + d.X, startPos.Y.Offset + d.Y)
                end
            end))
        end

                if isPointer(input) then
                    resizing, startIn, startSize = true, input.Position, main.Size
                    input.Changed:Connect(function()
                        if input.UserInputState == Enum.UserInputState.End then resizing = false end
                    end)
                end
            end)

                if resizing and isMove(input) then
                    local d = (input.Position - startIn) / uiScale.Scale
                    local vp = viewport()
                    local w = math.clamp(startSize.X.Offset + d.X, 540, math.max(540, vp.X / uiScale.Scale - 10))
                    local h = math.clamp(startSize.Y.Offset + d.Y, 360, math.max(360, vp.Y / uiScale.Scale - 10))
                    main.Size = UDim2.fromOffset(w, h)
                end
            end))

                local ping = 0

                statusLabel.Text = string.format("%s %s | FPS: %d | Ping: %d | %s",

            end
            refresh(0)
            track(RunService.RenderStepped:Connect(function()
                frames += 1
                local now = os.clock()
                if now - last >= 0.5 then
                    local fps = math.floor(frames / (now - last) + 0.5)
                    frames, last = 0, now
                    refresh(fps)
                end
            end))

            if not tab then return false end

                local on = (t == tab)
                t.page.Visible = on
                t.bar.Visible = on
                tween(t.title, 0.12, { TextColor3 = on and C.Text or C.Sub })
                tween(t.btn, 0.12, { BackgroundTransparency = on and 0.85 or 1 })

            pill.Text = tab.name
            return true
        end

                end

                local track_ = new("Frame", { BackgroundColor3 = C.ElemHi, Position = UDim2.new(0, 0, 0.5, -3), Size = UDim2.new(1, 0, 0, 6), BorderSizePixel = 0, Parent = hit })
                corner(track_, 3)
                local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0, Parent = track_ })
                accent(fill, "BackgroundColor3")
                corner(fill, 3)
                local knob = new("Frame", { BackgroundColor3 = Color3.fromRGB(255, 255, 255), AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(14, 14), Position = UDim2.new(0, 0, 0.5, 0), BorderSizePixel = 0, Parent = track_ })
                corner(knob, 7)
                local function paint()
                    local a = (max == min) and 0 or (value - min) / (max - min)
                    fill.Size = UDim2.new(a, 0, 1, 0)
                    knob.Position = UDim2.new(a, 0, 0.5, 0)
                    valLbl.Text = fmt(value)
                end
                paint()
                local dragging = false
                local function setFromX(x)
                    local w = track_.AbsoluteSize.X
                    if w <= 0 then return end
                    local a = math.clamp((x - track_.AbsolutePosition.X) / w, 0, 1)
                    local v = min + (max - min) * a
                    v = math.clamp(math.floor(v / step + 0.5) * step, min, max)
                    if v ~= value then
                        value = v
                        paint()
                        if type(c.Callback) == "function" then pcall(c.Callback, value) end
                    end
                end
                hit.InputBegan:Connect(function(input)
                    if isPointer(input) then
                        dragging = true
                        scroll.ScrollingEnabled = false
                        setFromX(input.Position.X)
                        input.Changed:Connect(function()
                            if input.UserInputState == Enum.UserInputState.End then
                                dragging = false
                                scroll.ScrollingEnabled = true
                            end
                        end)
                    end
                end)
                track(UIS.InputChanged:Connect(function(input)
                    if dragging and isMove(input) then setFromX(input.Position.X) end
                end))
                register(wrap, c.Title)
                local obj = {}
                function obj:Set(v) value = math.clamp(tonumber(v) or min, min, max) paint() end
                return obj
            end

            function adapter:Input(c)
                local bodyF = target(64)
                local wrap = new("Frame", { BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, Parent = bodyF })
                vlist(wrap, 4)
                local t = textBlock(wrap, c.Title, c.Desc)
                t.LayoutOrder = 1
                local box = new("TextBox", {
                    BackgroundColor3 = C.Elem, Size = UDim2.new(1, 0, 0, 34), PlaceholderText = tostring(c.Placeholder or "Enter text..."),
                    PlaceholderColor3 = C.Sub, Text = tostring(c.Value or c.Default or ""), TextColor3 = C.Text, TextSize = 14,
                    Font = FONT_MED, ClearTextOnFocus = false, BorderSizePixel = 0, LayoutOrder = 2, Parent = wrap,
                })
                corner(box, 6)
                stroke(box, C.Stroke, 1)
                box.FocusLost:Connect(function()
                    if type(c.Callback) == "function" then pcall(c.Callback, box.Text) end
                end)
                register(wrap, c.Title)
                local obj = {}
                function obj:Set(v) box.Text = tostring(v) end
                return obj
            end

            function adapter:Paragraph(c)
                local bodyF = target(60)
                local box = new("Frame", { BackgroundColor3 = C.Card, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BorderSizePixel = 0, Parent = bodyF })
                corner(box, 8)
                local st = stroke(box, C.Accent, 1)
                accent(st, "Color")
                padding(box, 10, 8, 10, 10)
                vlist(box, 3)
                if c.Title and c.Title ~= "" then
                    label(box, tostring(c.Title), 15, FONT_BOLD, C.Text, { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, TextWrapped = true, TextYAlignment = Enum.TextYAlignment.Top, LayoutOrder = 1 })
                end
                local desc = label(box, tostring(c.Desc or ""), 13, FONT_REG, C.Sub, { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, TextWrapped = true, RichText = true, TextYAlignment = Enum.TextYAlignment.Top, LayoutOrder = 2 })
                register(box, (c.Title or "") .. " " .. (c.Desc or ""))
                local obj = {}
                function obj:SetDesc(text) desc.Text = tostring(text or "") end
                return obj
            end

            if #tabs == 1 and not activeTab then selectTab(tab) end
            return adapter
        end

        function Window:Destroy()
            for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
            conns = {}
            accentList, hooks = {}, {}
            if toastHolder and toastHolder.Parent == gui then toastHolder = nil end
            pcall(function() gui:Destroy() end)
        end

        return Window
    end

    return Lib
end)()

local LoopDefs = {
    AutoRoll      = { delay = function() return 15 end,                  fn = doAutoRollAssert },
    ManualRoll    = { delay = function() return State.RollDelay end,     fn = doManualRoll },
    AutoCollect   = { delay = function() return State.CollectDelay end,  fn = doCollect },
    AutoEquipBest = { delay = function() return State.EquipBestDelay end,fn = doEquipBest },
    AutoRebirth   = { delay = function() return State.RebirthDelay end,  fn = doRebirth },
    AutoSell      = { delay = function() return State.SellDelay end,     fn = doSell },
    AutoRewards   = { delay = function() return State.RewardDelay end,   fn = doRewards },
    AutoSpin      = { delay = function() return State.SpinDelay end,     fn = doSpin },
    AutoTower     = { delay = function() return 15 end,                  fn = function() send("Towers", "SetAutoTower", true) end },
    AutoTowerTeam = { delay = function() return State.TowerTeamDelay end,fn = doTowerTeam },
    AutoReplay    = { delay = function() return State.ReplayDelay end,   fn = doReplay },
}

local function loopToggle(flag, extra)
    return function(v)
        local d = LoopDefs[flag]
        setLoop(flag, v, d.delay, d.fn)
        if extra then pcall(extra, v) end
    end
end

-- status + autosave thread
task.spawn(function()
    local tick = 0
    while Alive do
        tick = tick + 1
        local up = os.clock() - Stats.Started
        if StatusCard then
            pcall(function()
                StatusCard:SetDesc(("ออนไลน์: %02d:%02d:%02d\nคำสั่งที่ส่ง: %d | ผิดพลาด: %d | ทอย: %d\nLearn Mode: %s"):format(
                    math.floor(up / 3600), math.floor(up / 60) % 60, math.floor(up) % 60,
                    Stats.Calls, Stats.Errors, Stats.Rolls, LearnStatus))
            end)
        end
        if LearnCard then
            pcall(function()
                local lines = { ("สถานะ: %s\nจำได้แล้ว: %d คำสั่ง"):format(LearnStatus, #LearnOrder) }
                for i, l in ipairs(LearnLog) do lines[#lines + 1] = l if i >= 6 then break end end
                LearnCard:SetDesc(table.concat(lines, "\n"))
            end)
        end
        if State.SpeedEnabled and tick % 3 == 0 then pcall(applySpeed) end
        if dirty and tick % 5 == 0 then saveConfig() end
        task.wait(1)
    end
end)
