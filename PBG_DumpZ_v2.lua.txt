local function trim(value)
    local s = tostring(value or ""):gsub("^%s+", ""):gsub("%s+$", "")
    return s
end

local function safeFolder(path)
    if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder(path) then
        pcall(makefolder, path)
    end
end

safeFolder(ROOT_FOLDER)
safeFolder(DUMP_FOLDER)

local function loadJsonFile(path, fallback)
    if type(readfile) ~= "function" then return fallback end
    local okRead, body = pcall(readfile, path)
    if not okRead or type(body) ~= "string" or body == "" then return fallback end
    local okDecode, result = pcall(function() return HttpService:JSONDecode(body) end)
    if okDecode and type(result) == "table" then return result end
    return fallback
end

local function saveJsonFile(path, data)
    if type(writefile) ~= "function" then return false end
    safeFolder(ROOT_FOLDER)
    local okEncode, body = pcall(function() return HttpService:JSONEncode(data) end)
    if not okEncode then return false end
    return pcall(writefile, path, body)
end

local function safeProperty(object, property)
    local ok, value = pcall(function() return object[property] end)
    return ok, value
end

local function literal(value)
    local kind = typeof(value)
    if kind == "string" then return string.format("%q", value) end
    if kind == "number" or kind == "boolean" then return tostring(value) end
    if kind == "nil" then return "nil" end
    if kind == "Vector3" then return string.format("Vector3.new(%s, %s, %s)", value.X, value.Y, value.Z) end
    if kind == "Vector2" then return string.format("Vector2.new(%s, %s)", value.X, value.Y) end
    if kind == "Color3" then return string.format("Color3.new(%s, %s, %s)", value.R, value.G, value.B) end
    if kind == "CFrame" then
        local values = {value:GetComponents()}
        for i, n in ipairs(values) do values[i] = tostring(n) end
        return "CFrame.new(" .. table.concat(values, ", ") .. ")"
    end
    if kind == "EnumItem" then return tostring(value) end
    if kind == "Instance" then return string.format("%q", value:GetFullName()) end
    return string.format("%q", tostring(value))
end

local function resolveServiceTarget()
    local ok, target = pcall(function() return game:GetService(S.Service) end)
    if not ok or not target then return nil, S.Service end
    if S.SubPath ~= "" then
        for part in S.SubPath:gmatch("[^/]+") do
            target = target and target:FindFirstChild(part)
            if not target then break end
        end
    end
    return target, S.Service
end

local function dumpObject(root)
    local lines = {"Dump | " .. APP_NAME, "Root: " .. root:GetFullName(), "Class: " .. root.ClassName}
    local objects = {root}
    if S.Children then
        for _, object in ipairs(root:GetDescendants()) do objects[#objects + 1] = object end
    end
    for n, object in ipairs(objects) do
        lines[#lines + 1] = ""
        lines[#lines + 1] = "[" .. object.ClassName .. "] " .. object:GetFullName()
        if S.Attributes then
            local ok, attributes = pcall(function() return object:GetAttributes() end)
            if ok then
                for key, value in pairs(attributes) do
                    lines[#lines + 1] = "  @" .. tostring(key) .. " = " .. literal(value)
                end
            end
        end
        if S.Properties then
            for _, property in ipairs(PROPERTY_NAMES) do
                local ok, value = safeProperty(object, property)
                if ok then lines[#lines + 1] = "  ." .. property .. " = " .. literal(value) end
            end
        end
        yieldEvery(n, 300)
    end
    if S.SourceFormat then
        local source = {"-- " .. APP_NAME .. " source-format dump", "local dump = {}"}
        for n, object in ipairs(objects) do
            source[#source + 1] = string.format("dump[%q] = {ClassName = %q, Name = %q}", object:GetFullName(), object.ClassName, object.Name)
            yieldEvery(n, 300)
        end
        source[#source + 1] = "return dump"
        return table.concat(source, "\n")
    end
    return table.concat(lines, "\n")
end

local function performDump(saveToFile, showPreview)
    setText(UI.Status, L("status_working"))
    local target, serviceName = resolveServiceTarget()
    if not target then
        notify(APP_NAME, L("target_missing", tostring(serviceName)))
        setText(UI.Status, L("status_ready"))
        return
    end
    local ok, data = pcall(dumpObject, target)
    if not ok then
        notify(APP_NAME, L("dump_failed", tostring(data)))
        setText(UI.Status, L("status_ready"))
        return
    end
    LastDumpData = data
    if showPreview then
        local shown = data
        if #shown > PREVIEW_LIMIT then
            shown = shown:sub(1, PREVIEW_LIMIT) .. L("preview_cut", PREVIEW_LIMIT, #data)
        end
        setText(UI.Preview, shown)
        setText(UI.Status, L("status_preview", #data))
    end
    if saveToFile then
        safeFolder(DUMP_FOLDER)
        safeFolder(DUMP_FOLDER .. "/Dump")
        local name = S.FileName ~= "" and S.FileName or (serviceName .. "_" .. os.date("%Y%m%d_%H%M%S"))
        name = name:gsub("[^%w_%- ]", "_")
        local ext = S.SourceFormat and ".lua" or ".txt"
        LastDumpPath = DUMP_FOLDER .. "/Dump/" .. name .. ext
        if type(writefile) == "function" then pcall(writefile, LastDumpPath, data) end
        setText(UI.Status, L("status_saved", LastDumpPath))
        notify(APP_NAME, L("status_saved", LastDumpPath))
    end
end

--------------------------------------------------------------------
-- Remote Spy (ติดตั้ง hook ครั้งเดียว ใช้ได้ต่อเนื่องแม้เปลี่ยนภาษา)
--------------------------------------------------------------------
local Spy = {Enabled = false, Calls = {}, Skip = "sync,update,heartbeat,tick,ping,replicate,stream,position,character,camera,move,cframe", Installed = false, OnChange = nil}

local function shouldSkipRemote(name)
    name = tostring(name or ""):lower()
    for term in Spy.Skip:gmatch("[^,%s]+") do
        if name:find(term:lower(), 1, true) then return true end
    end
    return false
end

local function addSpyCall(remote, method, args)
    if not Spy.Enabled then return end
    local okName, name = pcall(function() return remote.Name end)
    if not okName or shouldSkipRemote(name) then return end
    local parts = {}
    for i = 1, args.n or #args do parts[#parts + 1] = literal(args[i]) end
    local okPath, path = pcall(function() return remote:GetFullName() end)
    Spy.Calls[#Spy.Calls + 1] = {Source = string.format("%s:%s(%s)", okPath and path or tostring(name), method, table.concat(parts, ", "))}
    while #Spy.Calls > 200 do table.remove(Spy.Calls, 1) end
    if Spy.OnChange then Spy.OnChange() end
end

local function installRemoteSpy()
    if Spy.Installed then return true end
    if type(hookmetamethod) ~= "function" or type(getnamecallmethod) ~= "function" then return false end
    local wrap = newcclosure or function(f) return f end
    local old
    old = hookmetamethod(game, "__namecall", wrap(function(self, ...)
        local method = getnamecallmethod()
        if Spy.Enabled and (method == "FireServer" or method == "InvokeServer") then
            local isRemote = false
            pcall(function()
                isRemote = typeof(self) == "Instance" and (self.ClassName == "RemoteEvent" or self.ClassName == "RemoteFunction")
            end)
            if isRemote and not (type(checkcaller) == "function" and checkcaller()) then
                local args = table.pack(...)
                task.defer(addSpyCall, self, method, args)
            end
        end
        return old(self, ...)
    end))
    Spy.Installed = true
    return true
end

local function refreshSpyUi()
    if not Spy.Enabled then
        setText(UI.SpyStatus, L("spy_off"))
    else
        setText(UI.SpyStatus, L("spy_on", #Spy.Calls))
    end
    if #Spy.Calls == 0 then
        setText(UI.SpyLog, L("spy_log_empty"))
        return
    end
    local lines = {}
    for i = math.max(1, #Spy.Calls - 14), #Spy.Calls do
        local src = Spy.Calls[i].Source
        if #src > 160 then src = src:sub(1, 160) .. "..." end
        lines[#lines + 1] = src
    end
    setText(UI.SpyLog, table.concat(lines, "\n"))
end

-- Map Dump (คงลำดับชั้น Parent/Child)
--------------------------------------------------------------------
local function generateMapScript()
    local rootName = S.MapFolder ~= "" and S.MapFolder or "ValenHub_Map"
    local lines = {
        "-- " .. APP_NAME .. " map dump",
        "-- PlaceId: " .. tostring(game.PlaceId),
        "-- Generated: " .. os.date("%Y-%m-%d %H:%M:%S"),
        "local function new(c) local ok, o = pcall(Instance.new, c) return ok and o or nil end",
        "local root = Instance.new(\"Folder\")",
        "root.Name = " .. string.format("%q", rootName),
        "root.Parent = workspace",
        "local refs = {}",
    }
    local skipChars = {}
    if not S.MapChars then
        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then skipChars[#skipChars + 1] = player.Character end
        end
    end
    local objects, index = {}, {}
    for n, object in ipairs(workspace:GetDescendants()) do
        local skip = false
        if object:IsA("Camera") or object:IsA("Terrain") then
            skip = true
        elseif not S.MapScripts and object:IsA("LuaSourceContainer") then
            skip = true
        elseif not S.MapChars and (object:IsA("Humanoid") or object:IsA("Animator")) then
            skip = true
        else
            for _, character in ipairs(skipChars) do
                if object == character or object:IsDescendantOf(character) then skip = true break end
            end
        end
        if not skip then
            objects[#objects + 1] = object
            index[object] = #objects
        end
        yieldEvery(n, 500)
    end
    for i, object in ipairs(objects) do
        local parent = object.Parent
        while parent and parent ~= workspace and not index[parent] do parent = parent.Parent end
        local parentExpr = (parent and index[parent]) and string.format("refs[%d] or root", index[parent]) or "root"
        local isPart = object:IsA("BasePart")
        lines[#lines + 1] = string.format("do local o = new(%q) if o then", object.ClassName)
        lines[#lines + 1] = string.format("o.Name = %q", object.Name)
        for _, property in ipairs(PROPERTY_NAMES) do
            if not (isPart and (property == "Position" or property == "Orientation")) then
                local ok, value = safeProperty(object, property)
                if ok and typeof(value) ~= "Instance" then
                    lines[#lines + 1] = string.format("pcall(function() o[%q] = %s end)", property, literal(value))
                end
            end
        end
        if S.MapScripts and object:IsA("LuaSourceContainer") then
            local ok, source = pcall(function() return object.Source end)
            if ok then lines[#lines + 1] = string.format("pcall(function() o.Source = %q end)", source) end
        end
        lines[#lines + 1] = string.format("o.Parent = %s", parentExpr)
        lines[#lines + 1] = string.format("refs[%d] = o end end", i)
        yieldEvery(i, 100)
    end
    lines[#lines + 1] = "return root"
    return table.concat(lines, "\n"), #objects
end
